Tag Shape Variants

These sections capture the exact top-level code-block tag combinations we tested.

Variant: tag_pre

def pattern_seven():
    rows = []
    for index in range(20):
        rows.append({
            "marker": "PATTERN-07-CODE",
            "index": index,
            "value": index * 3,
        })
    return rows

result = pattern_seven()
for row in result:
    print(row["marker"], row["index"], row["value"])
